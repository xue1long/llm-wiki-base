# src/vision/extractor.py
"""Image extraction from PDF (MVP); DOCX/PPTX/EPUB deferred to v2.0.1."""
from dataclasses import dataclass
from pathlib import Path


@dataclass
class ExtractedImage:
    task_id: str
    index: int
    bytes: bytes
    mime_type: str                # "image/png" | "image/jpeg"
    source_page: str              # "wiki/sources/<task_id>.md"
    context: str                  # surrounding text from PDF (for captioning)


def extract_from_pdf(pdf_path: Path, task_id: str) -> list[ExtractedImage]:
    """Extract images from PDF using pdfplumber."""
    import pdfplumber
    images: list[ExtractedImage] = []
    with pdfplumber.open(pdf_path) as pdf:
        for page_idx, page in enumerate(pdf.pages):
            for img_idx, img in enumerate(page.images):
                try:
                    # Get image bytes via pdfplumber
                    img_obj = img.to_image(resolution=150).original
                    # Get surrounding text as context
                    context_text = ""
                    for char in page.chars:
                        if img["top"] - 100 < char.get("top", 0) < img["bottom"] + 100:
                            context_text += char.get("text", "")
                    # Determine mime from format
                    fmt = img_obj.format.lower() if img_obj.format else "png"
                    mime = f"image/{fmt}" if fmt in ("png", "jpeg", "jpg") else "image/png"
                    from io import BytesIO
                    buf = BytesIO()
                    img_obj.save(buf, format=fmt)
                    images.append(ExtractedImage(
                        task_id=task_id,
                        index=len(images),
                        bytes=buf.getvalue(),
                        mime_type=mime,
                        source_page=f"wiki/sources/{task_id}.md",
                        context=context_text[:500],
                    ))
                except Exception:
                    continue  # skip broken image
    return images
```

# src/vision/captioner.py
"""LLM-based image captioning using vision-capable models."""
import asyncio
import base64
import logging

from ..llm.provider_factory import create_llm_provider
from ..llm.registry import ProviderRegistry
from .extractor import ExtractedImage


_logger = logging.getLogger(__name__)

MAX_IMAGES_PER_TASK = 20
MAX_CONCURRENT = 5


CAPTION_PROMPT = """Describe this image factually. Output strict JSON:
{
  "caption": "<1-3 sentences>",
  "alt_text": "<short alt, < 100 chars>",
  "entities": ["<named entities visible>"],
  "confidence": 0.0-1.0
}

Context (surrounding text):
{context}
"""


class ImageCaption:
    def __init__(self, task_id: str, index: int, caption: str, alt_text: str,
                 entities: list, confidence: float, model_used: str, generated_at: int):
        self.task_id = task_id
        self.index = index
        self.caption = caption
        self.alt_text = alt_text
        self.entities = entities
        self.confidence = confidence
        self.model_used = model_used
        self.generated_at = generated_at


class VisionCaptioner:
    def __init__(self, ctx, model: str = "gpt-4o-mini"):
        self.ctx = ctx
        self.model = model
        config = ProviderRegistry.get(ctx.settings.llm.provider_registry_name)
        self.provider = create_llm_provider(config.name, model_override=model)

    async def caption_one(self, image: ExtractedImage) -> ImageCaption:
        b64 = base64.b64encode(image.bytes).decode()
        prompt = CAPTION_PROMPT.format(context=image.context or "(no context)")
        try:
            response = await self.provider.complete(
                prompt=prompt,
                response_format={
                    "type": "object",
                    "properties": {
                        "caption": {"type": "string"},
                        "alt_text": {"type": "string"},
                        "entities": {"type": "array", "items": {"type": "string"}},
                        "confidence": {"type": "number"},
                    },
                    "required": ["caption", "alt_text"],
                },
            )
        except Exception as e:
            _logger.warning(f"[vision] caption failed: {e}")
            return ImageCaption(
                task_id=image.task_id, index=image.index,
                caption="[Caption generation failed]", alt_text="image",
                entities=[], confidence=0.0, model_used=self.model,
                generated_at=0,
            )
        import time
        return ImageCaption(
            task_id=image.task_id, index=image.index,
            caption=response.get("caption", ""),
            alt_text=response.get("alt_text", ""),
            entities=response.get("entities", []),
            confidence=response.get("confidence", 0.0),
            model_used=self.model,
            generated_at=int(time.time() * 1000),
        )

    async def caption_batch(self, images: list[ExtractedImage]) -> list[ImageCaption]:
        if len(images) > MAX_IMAGES_PER_TASK:
            _logger.warning(f"[vision] truncating {len(images)} to {MAX_IMAGES_PER_TASK}")
            images = images[:MAX_IMAGES_PER_TASK]
        sem = asyncio.Semaphore(MAX_CONCURRENT)
        async def cap(img):
            async with sem:
                return await self.caption_one(img)
        return await asyncio.gather(*[cap(i) for i in images])
```

# src/vision/storage.py
"""Store extracted images as MediaPage in wiki/media/."""
from pathlib import Path

from ..lib.write_hooks import safe_write
from ..wiki.types import PageType, WikiPage
from .captioner import ImageCaption
from .extractor import ExtractedImage


MEDIA_DIR = "wiki/media"


class MediaPage:
    @staticmethod
    def write(ctx, image: ExtractedImage, caption: ImageCaption) -> Path:
        media_dir = ctx.paths.wiki / "media"
        media_dir.mkdir(parents=True, exist_ok=True)
        ext = image.mime_type.split("/")[-1]
        image_path = media_dir / f"{image.task_id}_{image.index}.{ext}"
        image_path.write_bytes(image.bytes)

        # Write caption page
        page_id = f"{image.task_id}_{image.index}"
        page_path = media_dir / f"{page_id}.md"
        fm = {
            "id": page_id, "title": f"Image from {image.task_id}",
            "type": "media",
            "sources": [f"raw/sources/{image.task_id}.pdf"],
            "caption": caption.caption, "alt_text": caption.alt_text,
            "entities": caption.entities, "confidence": caption.confidence,
            "image": f"media/{image.task_id}_{image.index}.{ext}",
            "created_at": caption.generated_at, "updated_at": caption.generated_at,
        }
        from ..wiki.page_writer import write_page
        from ..wiki.page_writer import page_path_for
        # Save caption via WikiPage
        page = WikiPage(
            id=page_id, title=fm["title"], type=PageType.SOURCE,  # media pages reuse source for now
            sources=fm["sources"], created_at=fm["created_at"], updated_at=fm["updated_at"],
            body=f"![{caption.alt_text}](media/{image.task_id}_{image.index}.{ext})\n\n{caption.caption}",
        )
        write_page(ctx.paths, page)
        return page_path

```python
# src/cli_ext/vision_cmd.py
"""Vision CLI: list media pages, manually extract from a file."""
import argparse
import asyncio
import sys

from ..vision.extractor import extract_from_pdf
from ..vision.captioner import VisionCaptioner
from ..vision.storage import MediaPage
from ..project.context import ProjectContext, ProjectNotFoundError


def cmd_vision_list(args: argparse.Namespace) -> None:
    ctx = _resolve(args.project)
    from ..wiki.page_writer import read_page
    from ..wiki.paths import WikiPaths
    paths: WikiPaths = ctx.paths
    media_dir = paths.wiki / "media"
    if not media_dir.exists():
        print("No media pages.")
        return
    for f in sorted(media_dir.glob("*.md")):
        p = read_page(f)
        print(f"  {p.id}  {p.title}  heat={p.heat}")


def cmd_vision_extract(args: argparse.Namespace) -> None:
    """Manually extract + caption a PDF."""
    ctx = _resolve(args.project)
    pdf_path = Path(args.path)
    if not pdf_path.exists():
        print(f"File not found: {pdf_path}", file=sys.stderr); sys.exit(2)
    images = extract_from_pdf(pdf_path, task_id=args.task_id)
    print(f"Extracted {len(images)} images")
    if not images:
        return
    captioner = VisionCaptioner(ctx, model=args.model)
    captions = asyncio.run(captioner.caption_batch(images))
    for img, cap in zip(images, captions):
        MediaPage.write(ctx, img, cap)
    print(f"Wrote {len(captions)} media pages")


from pathlib import Path


def _resolve(project_id):
    try:
        return ProjectContext.resolve(project_id, by_id_only=True)
    except ProjectNotFoundError as e:
        print(f"Error: {e}", file=sys.stderr); sys.exit(2)
```
