**Goal:** 将 Wiki 规范文档（`docs/guides/wiki-spec.md`）作为 canonical source of truth，通过 pre-commit hook 触发同步脚本，自动更新 Generator 的 prompt 片段。

**Architecture:**
- `docs/guides/wiki-spec.md` — YAML frontmatter 存规范，Markdown body 存人类可读描述

- `src/pipeline/wiki_rules_prompt.py` — 由脚本生成的 prompt 片段，Generator 导入引用
- `.git/hooks/pre-commit` — 触发同步脚本

**Tech Stack:** Python 3.14, PyYAML, hashlib, pre-commit hook

# Wiki 规范

## ID 命名规则

ID 必须为小写 kebab-case slug，如 `shuang-dian`、`ceo-romance`。
不得使用 `index`、`log` 等保留 ID。

## Frontmatter 字段

### 必须字段

| 字段 | 类型 | 说明 |
|---|---|---|
| `id` | str | 与文件名一致，不含 `.md` |
| `title` | str | 显示标题 |
| `type` | PageType | `source` \| `entity` \| `concept` \| `synthesis` |

### 可选字段

`sources`、`relations`、`grade`、`processing_depth`、`is_immutable`、`heat`、`last_used_at`、`zombie_since`

## Body 规则

- 不得为空（空 body → LINT-EMPTY-BODY INFO 告警）
- 支持 Markdown：`**bold**`、`*italic*`、`## 标题`、`### 子标题`、`- 列表`
- 跨页引用使用 `[[slug]]` 语法，如 `[[shuang-dian]]`

## Task 2: 创建同步脚本 `scripts/sync_wiki_spec.py`

**Files:**
- Create: `scripts/sync_wiki_spec.py`
- Modify: `.gitignore`（加入 `.wiki-spec-md5`）

**Interfaces:**
- Produces: 更新 `src/pipeline/wiki_rules_prompt.py`（若 MD5 变化）

**Context:**
- 同步脚本是纯函数式：`read_spec_md5() → parse_frontmatter() → generate_prompt_py() → write_output()`
- 不依赖 ruflo-kb 内部模块，可独立测试
- MD5 文件用于检测规范文档是否变更

- [ ] **Step 1: 创建同步脚本**

```python
#!/usr/bin/env python3
"""Sync wiki-spec.md → src/pipeline/wiki_rules_prompt.py

Run via pre-commit hook. Compares MD5 of docs/guides/wiki-spec.md against
.wiki-spec-md5; if unchanged, exits 0 silently. If changed, regenerates
wiki_rules_prompt.py and updates the MD5 record.
"""
import hashlib
import sys
from pathlib import Path

import yaml

SPEC_PATH = Path("docs/guides/wiki-spec.md")
OUTPUT_PATH = Path("src/pipeline/wiki_rules_prompt.py")
MD5_PATH = Path(".wiki-spec-md5")


def _read_md5_record() -> str:
    return MD5_PATH.read_text().strip() if MD5_PATH.exists() else ""


def _write_md5_record(md5: str) -> None:
    MD5_PATH.write_text(md5 + "\n")


def _compute_md5(path: Path) -> str:
    return hashlib.md5(path.read_bytes()).hexdigest()


def _parse_frontmatter(spec_text: str) -> dict:
    _, fm = spec_text.split("---", 2)[1:]
    return yaml.safe_load(fm)


def _generate_prompt_module(data: dict) -> str:
    rules = data["rules"]
    id_rules = rules["id"]
    fm_rules = rules["frontmatter"]
    body_rules = rules["body"]

    lines = [
        '"""Auto-generated by scripts/sync_wiki_spec.py. Do not edit manually."""',
        "",
        "ID_RULES = {",
        f'    "pattern": "{id_rules["pattern"]}",',
        f'    "max_length": {id_rules["max_length"]},',
        f'    "reserved": {id_rules["reserved"]},',
        "}",
        "",
        "FRONTMATTER_RULES = {",
        f'    "required": {fm_rules["required"]},',
        f'    "optional": {fm_rules["optional"]},',
        "}",
        "",
        "BODY_RULES = {",
        f'    "min_length": {body_rules["min_length"]},',
        f'    "wikilink_syntax": "{body_rules["wikilink_syntax"]}",',
        f'    "allowed_markdown": {body_rules["allowed_markdown"]},',
        "}",
        "",
    ]

    summary_lines = [
        "WIKI_RULES_SUMMARY = '''",
        "## Wiki Page Rules",
        "- ID: kebab-case, max 64 chars, no reserved IDs",
        "- Frontmatter required: id, title, type",
        "- Body: non-empty, use [[slug]] for cross-references",
        "'''",
    ]
    lines.extend(summary_lines)

    return "\n".join(lines) + "\n"


def main() -> int:
    if not SPEC_PATH.exists():
        print(f"WARN: {SPEC_PATH} not found, skipping sync", file=sys.stderr)
        return 0

    current_md5 = _compute_md5(SPEC_PATH)
    last_md5 = _read_md5_record()

    if current_md5 == last_md5:
        return 0  # No change, nothing to do

    spec_text = SPEC_PATH.read_text(encoding="utf-8")
    data = _parse_frontmatter(spec_text)
    content = _generate_prompt_module(data)

    OUTPUT_PATH.parent.mkdir(parents=True, exist_ok=True)
    OUTPUT_PATH.write_text(content, encoding="utf-8")
    print(f"Generated {OUTPUT_PATH}")

    _write_md5_record(current_md5)

    # Stage the generated file so it gets committed along with spec changes
    import subprocess
    subprocess.run(["git", "add", str(OUTPUT_PATH)], check=True)

    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## Task 4: 安装 pre-commit hook

**Files:**
- Create: `scripts/setup_git_hooks.py`

**Context:**
- `setup_git_hooks.py` 是 idempotent 的，安装后每次 `git commit` 自动触发同步
- Hook 内容：调用 `scripts/sync_wiki_spec.py`，失败则 abort commit

- [ ] **Step 1: 创建 `scripts/setup_git_hooks.py`**

```python
#!/usr/bin/env python3
"""Install git hooks for this repository.

Creates .git/hooks/pre-commit that runs scripts/sync_wiki_spec.py.
Idempotent: safe to run multiple times.
"""
from pathlib import Path
import shutil
import sys

HOOK_CONTENT = """#!/usr/bin/env python3
# Auto-generated by scripts/setup_git_hooks.py
# Do not edit manually
import subprocess
import sys
sys.path.insert(0, "{cwd}")
sys.path.insert(0, "{repo_root}")
result = subprocess.run([sys.executable, "{sync_script}"], cwd="{cwd}")
sys.exit(result.returncode)
"""

REPO_ROOT = Path(__file__).parent.parent.resolve()
SYNC_SCRIPT = REPO_ROOT / "scripts" / "sync_wiki_spec.py"
CWD = REPO_ROOT
HOOK_PATH = REPO_ROOT / ".git" / "hooks" / "pre-commit"


def main() -> int:
    content = HOOK_CONTENT.format(
        cwd=str(CWD),
        repo_root=str(REPO_ROOT),
        sync_script=str(SYNC_SCRIPT),
    )
    HOOK_PATH.parent.mkdir(parents=True, exist_ok=True)
    HOOK_PATH.write_text(content, encoding="utf-8")
    HOOK_PATH.chmod(0o755)
    print(f"Installed pre-commit hook: {HOOK_PATH}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
